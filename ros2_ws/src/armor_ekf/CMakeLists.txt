cmake_minimum_required(VERSION 3.16)
project(armor_ekf)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(armor_interfaces REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d videoio)
find_package(Eigen3 REQUIRED)
get_filename_component(DEFAULT_ORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../deps/onnxruntime-linux-x64-1.30.0" ABSOLUTE)
set(ONNXRUNTIME_ROOT "${DEFAULT_ORT_ROOT}" CACHE PATH "ONNX Runtime C++ SDK directory")
find_path(ORT_INCLUDE onnxruntime_cxx_api.h PATHS "${ONNXRUNTIME_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
find_library(ORT_LIBRARY onnxruntime PATHS "${ONNXRUNTIME_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
add_library(armor_core src/pipeline.cpp src/reprojector.cpp)
target_include_directories(armor_core PUBLIC include ${OpenCV_INCLUDE_DIRS} ${ORT_INCLUDE})
target_link_libraries(armor_core PUBLIC Eigen3::Eigen ${OpenCV_LIBS} ${ORT_LIBRARY})
target_compile_options(armor_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(armor_ekf_node src/node.cpp)
target_link_libraries(armor_ekf_node PRIVATE armor_core
  rclcpp::rclcpp ${sensor_msgs_TARGETS} ${geometry_msgs_TARGETS}
  cv_bridge::cv_bridge ${armor_interfaces_TARGETS})
target_link_libraries(armor_ekf_node PRIVATE tf2_ros::tf2_ros ${visualization_msgs_TARGETS})
add_executable(armor_ekf_offline src/offline.cpp)
target_link_libraries(armor_ekf_offline PRIVATE armor_core)
add_executable(armor_ekf_replay src/replay.cpp)
target_link_libraries(armor_ekf_replay PRIVATE armor_core)
set_target_properties(armor_ekf_node armor_ekf_offline armor_ekf_replay PROPERTIES
  INSTALL_RPATH "${CMAKE_INSTALL_PREFIX}/lib/${PROJECT_NAME};${ONNXRUNTIME_ROOT}/lib")
install(TARGETS armor_core armor_ekf_node armor_ekf_offline armor_ekf_replay
  ARCHIVE DESTINATION lib/${PROJECT_NAME}
  LIBRARY DESTINATION lib/${PROJECT_NAME}
  RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(armor_core_test test/core_test.cpp)
  target_link_libraries(armor_core_test PRIVATE armor_core)
  add_test(NAME armor_core_test COMMAND armor_core_test)
  add_executable(armor_tracker_test test/tracker_test.cpp)
  target_link_libraries(armor_tracker_test PRIVATE armor_core)
  add_test(NAME armor_tracker_test COMMAND armor_tracker_test)
  add_executable(armor_world_test test/world_test.cpp)
  target_link_libraries(armor_world_test PRIVATE armor_core tf2::tf2)
  get_filename_component(REAL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
  set(ARMOR_TEST_MODEL "${REAL_SOURCE_DIR}/../../../01_神经网络/yolo_pose/final_model/best.onnx" CACHE FILEPATH "ONNX model for world-pipeline integration test")
  if(NOT EXISTS "${ARMOR_TEST_MODEL}")
    set(ARMOR_TEST_MODEL "${REAL_SOURCE_DIR}/../../../01_神经网络/yolo_pose/final_model/best.onnx" CACHE FILEPATH "ONNX model for world-pipeline integration test" FORCE)
  endif()
  add_test(NAME armor_world_test COMMAND armor_world_test "${ARMOR_TEST_MODEL}")
  add_executable(armor_reprojection_test test/reprojection_test.cpp)
  target_link_libraries(armor_reprojection_test PRIVATE armor_core)
  add_test(NAME armor_reprojection_test COMMAND armor_reprojection_test "${ARMOR_TEST_MODEL}")
endif()
ament_package()
